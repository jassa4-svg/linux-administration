#!/usr/bin/env bash
# luo_etusivu.sh – kirjoittaa Nginxin etusivun
# Ajo: sudo ./luo_etusivu.sh
set -euo pipefail


KOHDE="/var/www/html/index.html"

# 1. Tarkista, että skripti ajetaan root-oikeuksilla (EUID 0 = root)
if [[ $EUID -ne 0 ]]; then
  echo "Aja skripti sudolla: sudo $0" >&2
  exit 1
fi

# 2. Ota varmuuskopio vanhasta sivusta
if [[ -f "$KOHDE" ]]; then
  cp -- "$KOHDE" "$KOHDE.bak"
fi

# 3. Here-document: kaikki rivit "<<EOF" ja "EOF" välissä kirjoitetaan tiedostoon.
#    Koska EOF ei ole lainausmerkeissä, $(hostname) ja $(date) korvautuvat arvoillaan.
cat > "$KOHDE" <<EOF
<!DOCTYPE html>
<html lang="fi">
<head>
  <meta charset="UTF-8">
  <title>tervetuloa $(hostname)</title>
</head>
<body>
  <h1>Tervetuloa palvelimelle $(hostname)</h1>
  <p>Sivu päivitetty: $(date '+%d.%m.%Y %H:%M UTC %:::z')</p>
  <p>Sivu päivitetty: $(uptime -p)</p>
 <p>Sivu päivitetty: $(free -h | awk '/^Mem:/ {print "Total:", $2, "| Available:", $7}')<p>
<p>Toimii uudelleenkäynnistyksen jälkeen</p>

</body>
</html>
EOF

# 4. Nginx lukee tiedoston käyttäjänä www-data, joten muilla pitää olla lukuoikeus
chmod 644 "$KOHDE"

echo "Kirjoitettiin $KOHDE"
